TASK 3 — API CLIENT AND AUTH CONTEXT
───────────────────────────────────────────────────────────

Write: src/api/client.js

Axios instance with:
  baseURL: import.meta.env.VITE_API_BASE_URL
  withCredentials: true
    (sends httpOnly cookie with every request)

Request interceptor:
  Add Authorization: Bearer {access_token}
  from AuthContext to every request

Response interceptor:
  On 401 response:
    call POST /auth/refresh
    if refresh succeeds:
      update access_token in AuthContext
      retry original request with new token
    if refresh fails:
      clear AuthContext
      redirect to /login

Write: src/context/AuthContext.jsx

State:
  user: null | UserObject
  accessToken: null | string
  isLoading: bool (true during initial refresh)
  isAuthenticated: bool

On mount:
  call POST /auth/refresh automatically
  if success → set user + accessToken
  if fail → set isAuthenticated = false
  set isLoading = false either way

Functions:
  login(email, password) → void
    POST /auth/login
    set user + accessToken from response
    refresh_token set by backend cookie

  logout() → void
    POST /auth/logout
    clear user + accessToken
    redirect to /login

  refreshToken() → bool
    POST /auth/refresh
    update accessToken
    return true if success, false if fail

Write: src/hooks/useAuth.js
  const { user, isAuthenticated,
          isLoading, login, logout }
    = useAuth()
  Reads from AuthContext

Write: src/api/auth.js

  register(email, password, name)
  logout()
  refreshToken()
  getMe()
  updateMe(data)
  changePassword(current, newPass)
